# The Hunt (slice 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace daily trick training behind `[PLAY]` with a turn-based hunting RPG: five skills, one zone, four enemies with distinct behaviour, XP and levels.

**Architecture:** The rules live in a pure `combat.ts` that knows nothing about React or storage, with randomness injected as `rng: () => number` so any fight replays exactly in a test. Data modules describe skills, enemies and the zone; a persistence hook holds level/XP/clears; one panel component renders three views.

**Tech Stack:** React 19 + TypeScript, Vitest + Testing Library, localStorage. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-09-03-hunt-rpg-design.md`

## Global Constraints

- Plain ASCII only in all UI text, art and effects — no emoji (`CLAUDE.md`, "Visual theme").
- UI copy in English.
- Enemy art is small plain ASCII, three lines, not braille — it sits beside a much larger cat inside the existing panel.
- The cat's own art (`src/data/asciiCat.ts`) is not touched. Slice 3 handles appearance.
- Bracket-style button labels (`[ BACK ]`) except skill and zone options, which stay sentence-case.
- Randomness enters pure functions as an injected `rng: () => number`, never `Math.random()` inside them.
- New persisted state uses its own localStorage key, never `PetSave`.
- Coverage thresholds are enforced at 90% lines/statements/branches/functions (`vite.config.ts`).

```ts
export function xpToNext(level: number): number {
  return 100 + (level - 1) * 50
}

export function applyXp(level: number, xp: number, gained: number) {
  if (gained <= 0) return { level, xp, levelsGained: 0 }
  let nextLevel = level
  let pool = xp + gained
  let levelsGained = 0
  while (pool >= xpToNext(nextLevel)) {
    pool -= xpToNext(nextLevel)
    nextLevel += 1
    levelsGained += 1
  }
  return { level: nextLevel, xp: pool, levelsGained }
}
```

### Task 3: The combat engine

**Files:**
- Create: `src/data/combat.ts`
- Test: `src/data/combat.test.ts`

**Interfaces:**
- Consumes: `Skill`/`skillById` (Task 1), `Enemy`/`enemyById`/`FLEE_THRESHOLD` (Task 2).
- Produces:
  - `interface CombatState { enemyId: string; enemyHp: number; catHp: number; catMaxHp: number; sharpenTurns: number; guarding: boolean; enemyStunTurns: number; enemyWindingUp: boolean; cooldowns: Record<string, number>; log: string[]; outcome: 'ongoing' | 'won' | 'lost' | 'fled' }`
  - `startCombat(enemyId: string, catMaxHp: number): CombatState`
  - `takeTurn(state: CombatState, skillId: string, rng: () => number): CombatState`
  - `SHARPEN_BONUS = 2`, `SHARPEN_TURNS = 3`

`takeTurn` returns a new state and never mutates its argument. Turn order: the cat acts, the fight is checked for a win, the enemy acts unless stunned or fleeing, the fight is checked for a loss, then buffs and cooldowns tick down. A skill still on cooldown, or an unknown id, returns the state unchanged.

- [ ] **Step 1: Write the failing tests**

```ts
import { describe, expect, it } from 'vitest'
import { SHARPEN_BONUS, SHARPEN_TURNS, startCombat, takeTurn } from './combat'
import { enemyById } from './enemies'

const rngMin = () => 0
const rngMax = () => 0.999
// Feeds a scripted sequence, then repeats the last value.
function scripted(values: number[]) {
  let i = 0
  return () => values[Math.min(i++, values.length - 1)]
}

describe('startCombat', () => {
  it('starts both sides at full health with a clean slate', () => {
    const s = startCombat('crumb-beetle', 12)
    expect(s.catHp).toBe(12)
    expect(s.enemyHp).toBe(enemyById('crumb-beetle')!.maxHp)
    expect(s.outcome).toBe('ongoing')
    expect(s.sharpenTurns).toBe(0)
    expect(s.guarding).toBe(false)
    expect(s.cooldowns).toEqual({})
  })
})

describe('takeTurn', () => {
  it('does not mutate the state it is given', () => {
    const before = startCombat('crumb-beetle', 12)
    const snapshot = JSON.parse(JSON.stringify(before))
    takeTurn(before, 'swipe', rngMin)
    expect(before).toEqual(snapshot)
  })

  it('deals damage inside the skill range', () => {
    const start = startCombat('grasshopper', 12)
    const low = takeTurn(start, 'swipe', rngMin).enemyHp
    const high = takeTurn(start, 'swipe', rngMax).enemyHp
    expect(start.enemyHp - low).toBe(2)
    expect(start.enemyHp - high).toBe(4)
  })

  it('wins when the enemy runs out of health', () => {
    let s = startCombat('crumb-beetle', 20)
    while (s.outcome === 'ongoing') s = takeTurn(s, 'swipe', rngMax)
    expect(s.outcome).toBe('won')
    expect(s.enemyHp).toBe(0)
  })

  it('loses when the cat runs out of health', () => {
    let s = startCombat('pantry-rat', 4)
    while (s.outcome === 'ongoing') s = takeTurn(s, 'swipe', rngMax)
    expect(['lost', 'won']).toContain(s.outcome)
  })

  it('never reports negative health', () => {
    let s = startCombat('crumb-beetle', 3)
    for (let i = 0; i < 20 && s.outcome === 'ongoing'; i++) s = takeTurn(s, 'swipe', rngMax)
    expect(s.catHp).toBeGreaterThanOrEqual(0)
    expect(s.enemyHp).toBeGreaterThanOrEqual(0)
  })

  it('refuses an unknown skill without spending a turn', () => {
    const s = startCombat('crumb-beetle', 12)
    expect(takeTurn(s, 'nonsense', rngMin)).toBe(s)
  })

  describe('cooldowns', () => {
    it('puts a skill on cooldown after use', () => {
      const s = takeTurn(startCombat('pantry-rat', 20), 'pounce', rngMin)
      expect(s.cooldowns.pounce).toBeGreaterThan(0)
    })

    it('refuses a skill still on cooldown', () => {
      const first = takeTurn(startCombat('pantry-rat', 20), 'pounce', rngMin)
      expect(takeTurn(first, 'pounce', rngMin)).toBe(first)
    })

    it('frees the skill once the cooldown expires', () => {
      let s = takeTurn(startCombat('pantry-rat', 30), 'pounce', rngMin)
      const cd = s.cooldowns.pounce
      for (let i = 0; i < cd; i++) s = takeTurn(s, 'swipe', rngMin)
      expect(s.cooldowns.pounce ?? 0).toBe(0)
    })
  })

  describe('sharpen claws', () => {
    it('adds a flat bonus to the next attack', () => {
      const base = startCombat('pantry-rat', 30)
      const plain = base.enemyHp - takeTurn(base, 'swipe', rngMin).enemyHp
      const sharpened = takeTurn(base, 'sharpen', rngMin)
      const boosted = sharpened.enemyHp - takeTurn(sharpened, 'swipe', rngMin).enemyHp
      expect(boosted).toBe(plain + SHARPEN_BONUS)
    })

    it('lasts exactly the advertised number of turns', () => {
      let s = takeTurn(startCombat('pantry-rat', 40), 'sharpen', rngMin)
      expect(s.sharpenTurns).toBe(SHARPEN_TURNS)
      for (let i = 0; i < SHARPEN_TURNS; i++) s = takeTurn(s, 'swipe', rngMin)
      expect(s.sharpenTurns).toBe(0)
    })
  })

  describe('flatten ears', () => {
    it('negates exactly one incoming hit', () => {
      const guarded = takeTurn(startCombat('crumb-beetle', 20), 'guard', rngMax)
      expect(guarded.catHp).toBe(20)
      const after = takeTurn(guarded, 'swipe', rngMax)
      expect(after.catHp).toBeLessThan(20)
    })
  })

  describe('hiss', () => {
    it('makes the enemy miss its next turn', () => {
      const stunned = takeTurn(startCombat('crumb-beetle', 20), 'hiss', rngMax)
      expect(stunned.catHp).toBe(20)
      expect(stunned.enemyStunTurns).toBe(0) // spent on the turn it was applied
    })
  })

  describe('behaviour', () => {
    it('lets a wounded mouse flee', () => {
      let s = startCombat('house-mouse', 30)
      while (s.outcome === 'ongoing' && s.enemyHp > 1) s = takeTurn(s, 'swipe', rngMax)
      expect(['fled', 'won']).toContain(s.outcome)
    })

    it('does not let the boss flee', () => {
      let s = startCombat('pantry-rat', 200)
      while (s.outcome === 'ongoing') s = takeTurn(s, 'swipe', rngMax)
      expect(s.outcome).toBe('won')
    })

    it('has the grasshopper wind up before it strikes', () => {
      const first = takeTurn(startCombat('grasshopper', 30), 'swipe', rngMin)
      expect(first.catHp).toBe(30) // wind-up turn deals nothing
      expect(first.enemyWindingUp).toBe(true)
      const second = takeTurn(first, 'swipe', rngMin)
      expect(second.catHp).toBeLessThan(30)
    })

    it('has a plain enemy attack every turn', () => {
      const s = takeTurn(startCombat('crumb-beetle', 30), 'swipe', rngMin)
      expect(s.catHp).toBeLessThan(30)
    })
  })

  it('writes a line to the log for every turn', () => {
    const s = takeTurn(startCombat('crumb-beetle', 20), 'swipe', rngMin)
    expect(s.log.length).toBeGreaterThan(0)
    for (const line of s.log) expect(line).toMatch(/^[\x20-\x7E]*$/)
  })

  it('ignores further turns once the fight is over', () => {
    let s = startCombat('crumb-beetle', 40)
    while (s.outcome === 'ongoing') s = takeTurn(s, 'swipe', rngMax)
    expect(takeTurn(s, 'swipe', rngMax)).toBe(s)
  })

  it('is deterministic for the same scripted rolls', () => {
    const script = () => scripted([0.1, 0.6, 0.3, 0.9, 0.2])
    const runA = takeTurn(takeTurn(startCombat('grasshopper', 20), 'swipe', script()), 'swipe', script())
    const runB = takeTurn(takeTurn(startCombat('grasshopper', 20), 'swipe', script()), 'swipe', script())
    expect(runA).toEqual(runB)
  })
})
```
